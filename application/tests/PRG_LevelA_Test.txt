(* ============================================================
   PRG_LevelA_Test — E_PROGRESS projection test (Level A)
   SML v5 — new

   Verifies, WITHOUT hardware, that the eProgress output introduced in
   Level A follows the expected transitions. It drives SML_Power with a
   small inline CiA402 emulator (StatusWord derived from ControlWord)
   and records the observed E_PROGRESS values.

   Expected sequence on Power.eProgress:
     idle            -> PROGRESS_INVALID
     enabling        -> PROGRESS_BUSY (transient, a few cycles)
     OperationEnabled-> PROGRESS_DONE
     injected fault  -> PROGRESS_ERROR
     drop execute    -> PROGRESS_INVALID

   RESULT: xTestPassed = TRUE when all four values have been observed.
   No dependency beyond OpenSML_Axis, SML_Power, E_PROGRESS and TON
   (Tc2_Standard).

   NOTE: this is a development test bench, not part of the library.
   ============================================================ *)

PROGRAM PRG_LevelA_Test
VAR
    Axis  : OpenSML_Axis;   // struct shared app <-> FB (no I/O mapping)
    Power : SML_Power;      // FB under test

    // ── test sequencer ───────────────────────────────────────
    step  : INT;
    tStep : TON;

    // ── stimuli ──────────────────────────────────────────────
    xEnable      : BOOL;    // drive enable request
    xInjectFault : BOOL;    // inject a fault (StatusWord bit 3)

    // ── observation ──────────────────────────────────────────
    ePwr : E_PROGRESS;      // current eProgress of Power

    xSeenInvalid : BOOL;    // flag "value observed at least once"
    xSeenBusy    : BOOL;
    xSeenDone    : BOOL;
    xSeenError   : BOOL;
    xTestPassed  : BOOL;    // TRUE when the full sequence has been seen
    xTestDone    : BOOL;    // sequence finished
END_VAR

// ─────────────────────────────────────────────────────────────
//  Sequencer: ~200 ms per step
// ─────────────────────────────────────────────────────────────
tStep(IN := NOT xTestDone, PT := T#200MS);
IF tStep.Q THEN
    tStep(IN := FALSE);
    step := step + 1;
END_IF

CASE step OF
    0:  // idle -> expected PROGRESS_INVALID
        xEnable      := FALSE;
        xInjectFault := FALSE;

    1:  // enable request -> BUSY (transient) then DONE
        xEnable := TRUE;

    // 2..4: holds enable, expected PROGRESS_DONE

    5:  // inject fault -> PROGRESS_ERROR (Power state 50 -> 999)
        xInjectFault := TRUE;

    6:  // release enable and fault -> back to PROGRESS_INVALID
        xEnable      := FALSE;
        xInjectFault := FALSE;

    7:  // end
        xTestDone := TRUE;
END_CASE

// ─────────────────────────────────────────────────────────────
//  FB under test
// ─────────────────────────────────────────────────────────────
Power(
    Axis           := Axis,
    xExecute       := xEnable,
    xEmergencyStop := TRUE);       // 1 = run (no quick-stop)

ePwr := Power.eProgress;

// ─────────────────────────────────────────────────────────────
//  Mini CiA402 emulator (called AFTER Power: it responds to the
//  ControlWord written by Power in the current cycle)
//    bit1 EnableVoltage  -> StatusWord.0 ReadyToSwitchOn
//    bit0 SwitchOn       -> StatusWord.1 SwitchedOn (requires voltage)
//    bit3 EnableOp       -> StatusWord.2 OperationEnabled (requires SwitchedOn)
//    bit7 FaultReset     -> clears StatusWord.3
// ─────────────────────────────────────────────────────────────
IF xInjectFault THEN
    Axis.StatusWord.3 := TRUE;                 // Fault
ELSIF Axis.ControlWord.7 THEN
    Axis.StatusWord.3 := FALSE;                // Fault Reset
END_IF

Axis.StatusWord.0 := Axis.ControlWord.1;                          // ReadyToSwitchOn
Axis.StatusWord.1 := Axis.ControlWord.0 AND Axis.ControlWord.1;   // SwitchedOn
Axis.StatusWord.2 := Axis.ControlWord.3 AND Axis.StatusWord.1;    // OperationEnabled
Axis.StatusWord.5 := TRUE;                                        // QuickStop not active

// ─────────────────────────────────────────────────────────────
//  Result collection
// ─────────────────────────────────────────────────────────────
CASE ePwr OF
    PROGRESS_INVALID: xSeenInvalid := TRUE;
    PROGRESS_BUSY:    xSeenBusy    := TRUE;
    PROGRESS_DONE:    xSeenDone    := TRUE;
    PROGRESS_ERROR:   xSeenError   := TRUE;
END_CASE

xTestPassed := xSeenInvalid AND xSeenBusy AND xSeenDone AND xSeenError;
