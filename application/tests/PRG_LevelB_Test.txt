(* ============================================================
   PRG_LevelB_Test - FB_AxisCtrl test (Level B)
   SML v6 - new

   Drives FB_AxisCtrl in xSimulation with a small inline CiA402
   emulator (StatusWord derived from ControlWord) through a
   sequence of eCmd commands, and verifies:
     - command routing to the leaf FBs,
     - the expected State.eProgress values,
     - decomposition of the COMBINED STATE with f_GetState /
       f_GetProgress (Level A functions, now used).

   Sequence:  ENABLE -> MOVE_CSP -> JOG_POS -> STOP -> DISABLE
   Expected:  ENABLE->DONE, CSP->DONE, JOG->BUSY, STOP->DONE

   RESULT: xTestPassed = TRUE when all expectations are observed.
   Development test bench, not part of the library.
   ============================================================ *)

PROGRAM PRG_LevelB_Test
VAR
    // -- data contract + FB under test ------------------------
    Axis  : OpenSML_Axis;
    Ctrl  : ST_AXIS_CTRL;
    Data  : ST_MOVE_DATA;
    State : ST_AXIS_STATE;
    Info  : ST_AXIS_INFO;
    Ax    : FB_AxisCtrl;

    // -- sequencer --------------------------------------------
    step  : INT;
    tStep : TON;

    // -- observation ------------------------------------------
    eProg   : E_PROGRESS;   // State.eProgress
    eStBase : DINT;         // f_GetState(State.eState)
    eStProg : E_PROGRESS;   // f_GetProgress(State.eState)

    xSeenEnableDone : BOOL;
    xSeenCspDone    : BOOL;
    xSeenJogBusy    : BOOL;
    xSeenStopDone   : BOOL;
    xTestPassed     : BOOL;
    xTestDone       : BOOL;
END_VAR

// -- configuration (every cycle) ------------------------------
Ctrl.xSimulation    := TRUE;
Ctrl.xEmergencyStop := TRUE;
Ctrl.Scale          := 1.0;      // 1 count/unit for simplicity
Ctrl.CycleTime      := 0.001;
Ctrl.PositionWindow := 0.5;
Data.lrAcceleration := 1000.0;
Data.lrDeceleration := 1000.0;
Data.lrJerk         := 10000.0;

// -- sequencer ~300 ms/step -----------------------------------
tStep(IN := NOT xTestDone, PT := T#300MS);
IF tStep.Q THEN
    tStep(IN := FALSE);
    step := step + 1;
END_IF

CASE step OF
    0:  Ctrl.eCmd := AXIS_ENABLE;                                  // -> DONE
    2:  Data.lrTargetPosition := 50.0;
        Data.lrVelocity       := 100.0;
        Ctrl.eCmd := AXIS_MOVE_CSP;                                // -> DONE (xInPosition)
    5:  Data.lrVelocity := 20.0;
        Ctrl.eCmd := AXIS_JOG_POS;                                 // -> BUSY (continuous)
    7:  Ctrl.eCmd := AXIS_STOP;                                    // -> DONE (standstill)
    9:  Ctrl.eCmd := AXIS_DISABLE;                                 // -> DONE
    11: xTestDone := TRUE;
END_CASE

// -- FB under test --------------------------------------------
Ax(Axis := Axis, Ctrl := Ctrl, Data := Data, State := State, Info := Info);

// -- mini CiA402 emulator (after the FB) ----------------------
Axis.Modes_of_operation_display := Axis.Modes_of_operation;      // the drive echoes the mode
Axis.StatusWord.0  := Axis.ControlWord.1;                        // ReadyToSwitchOn
Axis.StatusWord.1  := Axis.ControlWord.0 AND Axis.ControlWord.1; // SwitchedOn
Axis.StatusWord.2  := Axis.ControlWord.3 AND Axis.StatusWord.1;  // OperationEnabled
Axis.StatusWord.5  := TRUE;                                      // QuickStop not active
Axis.StatusWord.10 := Axis.ControlWord.8;                        // standstill when Halt

// -- observation: combined-state decomposition ---------------
eProg   := State.eProgress;
eStBase := f_GetState(State.eState);      // functional base (E_AXIS_STATE)
eStProg := f_GetProgress(State.eState);   // progress (E_PROGRESS)

IF (State.eCmdActive = AXIS_ENABLE)   AND (eProg = PROGRESS_DONE) THEN xSeenEnableDone := TRUE; END_IF
IF (State.eCmdActive = AXIS_MOVE_CSP) AND (eProg = PROGRESS_DONE) THEN xSeenCspDone    := TRUE; END_IF
IF (State.eCmdActive = AXIS_JOG_POS)  AND (eProg = PROGRESS_BUSY) THEN xSeenJogBusy    := TRUE; END_IF
IF (State.eCmdActive = AXIS_STOP)     AND (eProg = PROGRESS_DONE) THEN xSeenStopDone   := TRUE; END_IF

xTestPassed := xSeenEnableDone AND xSeenCspDone AND xSeenJogBusy AND xSeenStopDone;
