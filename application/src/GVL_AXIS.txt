(* ============================================================
   GVL_AXIS — Per-axis arrays of the SML Motion Layer
   SML v7 — Level B (new)

   Equivalent to GVL_AXIS of PLC_MOTION_LAYER, but:
     - the execution level is FB_AxisCtrl (SML leaf FBs),
       not FB_McAxisCtrl (MC_*);
     - wiring is via VAR_IN_OUT (no pointer array):
       MAIN passes Axis/Ctrl/Data/State/Info[nAxis] in the call.

   Axis[]  : drive PDO images; link to the I/O with SML_TC3Link.
   Ctrl[]/Data[]  : command + setpoint  (app -> layer)
   State[]/Info[] : result + actual values (layer -> app)
   Control[]      : per-axis control FB
   ItfSmlAxis[]   : interface access for high-level coordinators
   ============================================================ *)

{attribute 'qualified_only'}
VAR_GLOBAL
    Axis       : ARRAY[1..GVL_App.MAX_AXIS] OF OpenSML_Axis;    // PDO image (I/O link)
    Ctrl       : ARRAY[1..GVL_App.MAX_AXIS] OF ST_AXIS_CTRL;    // command + config
    Data       : ARRAY[1..GVL_App.MAX_AXIS] OF ST_MOVE_DATA;    // motion setpoint
    State      : ARRAY[1..GVL_App.MAX_AXIS] OF ST_AXIS_STATE;   // result + combined state
    Info       : ARRAY[1..GVL_App.MAX_AXIS] OF ST_AXIS_INFO;    // actual values

    Control    : ARRAY[1..GVL_App.MAX_AXIS] OF FB_AxisCtrl;  // per-axis control FB
    ItfSmlAxis : ARRAY[1..GVL_App.MAX_AXIS] OF I_Axis;       // interface access (I_Axis)
END_VAR
