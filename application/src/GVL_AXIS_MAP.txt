(* ============================================================
   GVL_AXIS_MAP - Bus side of the Motion Layer (per-axis UNION)
   SML v8 - Level B / MAPPING (new)

   Array of UNIONs to link to the fieldbus process image (or to
   expose via ADS). FB_Mapping_In/Out copy between these and GVL_AXIS.

   Typical direction (from the PLC's point of view):
     Ctrl[] , Data[]   : INPUTS   (command from outside)  -> %I* / RD
     State[], Info[]   : OUTPUTS  (state toward outside)   -> %Q* / WR

   Each element is a UNION: linkable as a struct (.stData) or as a
   raw block (.aRaw). Always present; MAPPING is active only if
   AXIS_MAP_ENABLE = TRUE (runtime flag; no conditional-compile
   pragma, not portable in the declaration part on CoDeSys).
   ============================================================ *)

{attribute 'qualified_only'}
VAR_GLOBAL
    // MAPPING enable flag (design-time or runtime)
    AXIS_MAP_ENABLE : BOOL := FALSE;   // TRUE = activate FB_Mapping_In / FB_Mapping_Out

    Ctrl  : ARRAY[1..GVL_App.MAX_AXIS] OF U_AXIS_CTRL;    // command   (bus -> plc)
    Data  : ARRAY[1..GVL_App.MAX_AXIS] OF U_MOVE_DATA;    // setpoint  (bus -> plc)
    State : ARRAY[1..GVL_App.MAX_AXIS] OF U_AXIS_STATE;   // state     (plc -> bus)
    Info  : ARRAY[1..GVL_App.MAX_AXIS] OF U_AXIS_INFO;    // info      (plc -> bus)
END_VAR
