(* ============================================================
   PRG_Mapping_In — Copy command/setpoint from the bus into GVL_AXIS
   SML v8 — Level B / MAPPING (new)

   Path: GVL_AXIS_MAP.Ctrl/Data (fieldbus)  ->  GVL_AXIS.Ctrl/Data
   Call at the START of MAIN, before the axis loop.

   GATE: active only if GVL_AXIS_MAP.AXIS_MAP_ENABLE = TRUE; otherwise
   an immediate bypass (behavior = v7). A runtime flag instead of a
   {IF defined(...)} pragma: the latter is not portable in the
   declaration part on CoDeSys. For COMPILE-TIME exclusion, as an
   alternative, wrap the BODY in {IF defined (AXIS_MAP)} ... {END_IF}
   in the IMPLEMENTATION part (not in the declaration).

   Runtime size check: if a struct exceeds the bus block (MAP_SIZE_*),
   the mapping is blocked for safety.

   Copy via structured assignment (portable, no dependency on a memcpy
   library). To link the RAW view use .aRaw on the fieldbus: the
   .stData copy stays correct.
   ============================================================ *)

PROGRAM PRG_Mapping_In
VAR
    i       : UINT;
    bSizeOk : BOOL;
END_VAR

IF NOT GVL_AXIS_MAP.AXIS_MAP_ENABLE THEN RETURN; END_IF

// ── Bus block size check (safety) ────────────────────────────
bSizeOk := (SIZEOF(ST_AXIS_CTRL) <= GVL_SML_CONST.MAP_SIZE_CTRL)
       AND (SIZEOF(ST_MOVE_DATA) <= GVL_SML_CONST.MAP_SIZE_DATA);
IF NOT bSizeOk THEN RETURN; END_IF   // blocked: resize MAP_SIZE_*

// ── Copy bus -> internal ─────────────────────────────────────
FOR i := 1 TO GVL_App.MAX_AXIS DO
    GVL_AXIS.Ctrl[i] := GVL_AXIS_MAP.Ctrl[i].stData;
    GVL_AXIS.Data[i] := GVL_AXIS_MAP.Data[i].stData;
END_FOR
