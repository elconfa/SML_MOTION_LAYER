(* ============================================================
   GVL_IO - I/O bridge process images (hardware)
   SML v9 - I/O bridge

   Used ONLY with the bridge. On CoDeSys with a direct link it is NOT
   needed: in that case leave IO_LINK_ENABLE = FALSE (the two
   FB_IoLink_* auto-bypass) and map GVL_AXIS.Axis[n] directly.

   MAPPING (important): DriveOut/DriveIn are PLAIN arrays, without
   'AT %Q*'/'AT %I*'. Reason: on CoDeSys an incomplete address in a
   GVL requires a VAR_CONFIG/mapping and otherwise throws error C0128.
   With plain arrays the code compiles standalone; the PDO channels
   are linked like this:
     - CoDeSys: <Slave> I/O Mapping tab -> Variable column ->
       bind each PDO to GVL_IO.DriveOut[n].<field> / DriveIn[n].<field>.
     - TwinCAT: 'Change Link' on each PDO to GVL_IO.DriveOut/DriveIn,
       or (if you prefer the explicit process image) re-add
       'AT %Q*' / 'AT %I*' - supported on TwinCAT.
   See docs/GUIDA_IO_Linking.md.
   ============================================================ *)

{attribute 'qualified_only'}
VAR_GLOBAL
    // bridge enable flag (design-time or runtime)
    IO_LINK_ENABLE : BOOL := FALSE;   // TRUE = activate FB_IoLink_In / _Out

    DriveOut : ARRAY[1..GVL_App.MAX_AXIS] OF ST_DriveOut;  // -> RxPDO (map in the configurator)
    DriveIn  : ARRAY[1..GVL_App.MAX_AXIS] OF ST_DriveIn;   // <- TxPDO (map in the configurator)
END_VAR
