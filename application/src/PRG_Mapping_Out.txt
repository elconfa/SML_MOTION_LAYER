(* ============================================================
   PRG_Mapping_Out — Copy state/info from GVL_AXIS toward the bus
   SML v8 — Level B / MAPPING (new)

   Path: GVL_AXIS.State/Info  ->  GVL_AXIS_MAP.State/Info (fieldbus)
   Call at the END of MAIN, after the axis loop.

   GATE: active only if GVL_AXIS_MAP.AXIS_MAP_ENABLE = TRUE; otherwise
   bypass (runtime flag; see the note in PRG_Mapping_In).
   ============================================================ *)

PROGRAM PRG_Mapping_Out
VAR
    i       : UINT;
    bSizeOk : BOOL;
END_VAR

IF NOT GVL_AXIS_MAP.AXIS_MAP_ENABLE THEN RETURN; END_IF

// ── Bus block size check (safety) ────────────────────────────
bSizeOk := (SIZEOF(ST_AXIS_STATE) <= GVL_SML_CONST.MAP_SIZE_STATE)
       AND (SIZEOF(ST_AXIS_INFO)  <= GVL_SML_CONST.MAP_SIZE_INFO);
IF NOT bSizeOk THEN RETURN; END_IF   // blocked: resize MAP_SIZE_*

// ── Copy internal -> bus ─────────────────────────────────────
FOR i := 1 TO GVL_App.MAX_AXIS DO
    GVL_AXIS_MAP.State[i].stData := GVL_AXIS.State[i];
    GVL_AXIS_MAP.Info[i].stData  := GVL_AXIS.Info[i];
END_FOR
