(* ============================================================
   FB_AxisCycleDemo - Example application using SML
   SML v9 - example (NOT part of the library)

   Shows the typical "command -> wait for State.xDone" pattern on an
   axis handled by the Motion Layer. It runs a continuous cycle:

     ENABLE -> HOME (once) -> A -> dwell -> B -> dwell -> A -> ...

   until xStop; on error -> automatic RESET and return to idle.

   It commands the axis via GVL_AXIS[nAxis] (writes Ctrl/Data, reads
   State/Info). It is the reference example for a programmer
   integrating SML into a real machine.

   NOTE on re-triggering PP (discrete) moves: between two AXIS_MOVE_ABS
   it goes through AXIS_NULL (the dwell steps) to create the edge that
   restarts the next move. For smooth re-targeting use AXIS_MOVE_CSP
   instead.
   ============================================================ *)

FUNCTION_BLOCK FB_AxisCycleDemo
VAR_INPUT
    nAxis   : UINT  := 1;          // axis index in GVL_AXIS [1..MAX_AXIS]
    xStart  : BOOL;                // start the cycle
    xStop   : BOOL;                // stop the cycle (Halt + end)
    lrPosA  : LREAL := 0.0;        // position A [units]
    lrPosB  : LREAL := 100.0;      // position B [units]
    lrVel   : LREAL := 100.0;      // move velocity [units/s]
    tDwell  : TIME  := T#500MS;    // dwell at each end
END_VAR
VAR_OUTPUT
    xBusy   : BOOL;                // cycle running
    xDone   : BOOL;                // cycle finished (after xStop)
    xError  : BOOL;                // axis error
    iStep   : INT;                 // current step (diagnostics)
    sDiag   : STRING(80);          // axis diagnostic text
END_VAR
VAR
    tonDwell : TON;
END_VAR

// -- mirror of axis state -------------------------------------
xError := GVL_AXIS.State[nAxis].xError;
sDiag  := GVL_AXIS.State[nAxis].DiagText;

// dwell timer (active only in steps 40/60, auto-resets on exit)
tonDwell(IN := (iStep = 40) OR (iStep = 60), PT := tDwell);

// -- supervision: error or stop while in the cycle ------------
IF (iStep > 0) AND (iStep < 900) THEN
    IF xError    THEN iStep := 900; END_IF
    IF xStop     THEN iStep := 200; END_IF
END_IF

CASE iStep OF

    0: // IDLE - axis stopped, enable kept if already on
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_NULL;
        xBusy := FALSE;
        IF xStart THEN
            xDone := FALSE;
            iStep := 10;
        END_IF

    10: // ENABLE - wait for OperationEnabled
        xBusy := TRUE;
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_ENABLE;
        IF GVL_AXIS.Info[nAxis].xEnabled THEN
            IF GVL_AXIS.Info[nAxis].xHomed THEN
                iStep := 30;         // already homed: skip homing
            ELSE
                iStep := 20;
            END_IF
        END_IF

    20: // HOME (once)
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_HOME;
        IF GVL_AXIS.State[nAxis].xDone THEN iStep := 30; END_IF

    30: // MOVE -> A
        GVL_AXIS.Data[nAxis].lrTargetPosition := lrPosA;
        GVL_AXIS.Data[nAxis].lrVelocity       := lrVel;
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_MOVE_ABS;
        IF GVL_AXIS.State[nAxis].xDone THEN iStep := 40; END_IF

    40: // dwell at A (NULL creates the edge for the next move)
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_NULL;
        IF tonDwell.Q THEN iStep := 50; END_IF

    50: // MOVE -> B
        GVL_AXIS.Data[nAxis].lrTargetPosition := lrPosB;
        GVL_AXIS.Data[nAxis].lrVelocity       := lrVel;
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_MOVE_ABS;
        IF GVL_AXIS.State[nAxis].xDone THEN iStep := 60; END_IF

    60: // dwell at B, then repeat the cycle (back to A)
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_NULL;
        IF tonDwell.Q THEN iStep := 30; END_IF

    200: // STOP requested - Halt and end
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_STOP;
        IF NOT GVL_AXIS.Info[nAxis].xMoving THEN
            GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_NULL;
            xBusy := FALSE;
            xDone := TRUE;
            iStep := 0;
        END_IF

    900: // ERROR - automatic reset, then idle (needs a new xStart)
        GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_RESET;
        xBusy := FALSE;
        IF NOT xError THEN
            GVL_AXIS.Ctrl[nAxis].eCmd := AXIS_NULL;
            iStep := 0;
        END_IF

END_CASE
