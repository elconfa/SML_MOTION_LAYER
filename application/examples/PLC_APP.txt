(* ============================================================
   PLC_APP — 2-axis machine application (SML usage example)
   SML v9 — example (NOT part of the library)

   Piece measuring/sorting machine on a belt:
     - Axis 1 (nBelt) = continuously rotating belt (velocity), with TouchProbe
     - Axis 2 (nPos)  = positioner

   CYCLE:
     HOMING (once): axis1 homes in place; axis2 homes on a sensor+encoder
       index with an offset -> reference = lrAx2Home (e.g. 10 mm).
     Then at steady state:
       1. axis1 starts in velocity + arms the TouchProbe.
       2. a piece covers the photocell -> TouchProbe latches (rising); the belt
          switches WITHOUT stopping from velocity to relative Profile Position
          lrBeltMove (e.g. 500 mm).
       3. the piece clears the photocell -> TouchProbe latches (falling); the
          TouchProbe is stopped and the measurement = |falling - rising| / Scale.
       4a. measurement > threshold (lrThreshold): the belt KEEPS running in velocity;
           the piece, passing under the exit photocell (xExitPhotocell), is
           certified as ejected (it falls to the ground). The cycle restarts (no homing).
       4b. measurement <= threshold: wait for the belt (PP) to stop, a short wait,
           axis2 goes FORWARD to lrAx2Fwd absolute (e.g. +400 mm), a short wait, then
           SIMULTANEOUSLY axis2 returns to lrAx2Zero (0 mm) and axis1 restarts in
           velocity. The cycle restarts.

   PIPELINING: a reject does NOT stop the belt; while the rejected piece travels
   toward the exit photocell (counted in parallel by a tracker on
   xExitPhotocell), a NEW piece can already enter and be measured.

   STANDBY: if no piece arrives for tIdle, the belt slows and stops
   (xStandby). While stopped it can be re-homed with xHomeBelt; the external
   signal xResume reactivates rotation (exit from standby).

   INTEGRATION: input commands (VAR_INPUT) and output state (VAR_OUTPUT).
   Put PLC_APP in the task TOGETHER with MAIN (which runs the axes).

   ASSUMPTIONS (adaptable via parameters):
     - the measure photocell is wired to the TouchProbe trigger of axis1's drive AND
       as the PLC input xPhotocell (TRUE = covered);
     - the exit photocell (xExitPhotocell) downstream: certifies the ejection of
       rejected pieces (no actuator, the piece falls off the belt);
     - homing of the two axes performed by the drive's homing method;
     - HomeOffset in ENCODER COUNTS (converted from lrAx2Home * lrScale);
     - axis2 moves (forward to lrAx2Fwd, return to lrAx2Zero) ABSOLUTE.
   ============================================================ *)

PROGRAM PLC_APP
VAR_INPUT
    // ── machine commands ─────────────────────────────────────
    xStart      : BOOL;                 // TRUE = start/hold the automatic cycle
    xStop       : BOOL;                 // TRUE = stop the cycle (Halt) and go to idle
    xReset      : BOOL;                 // (reserved) external reset
    xSafetyOk   : BOOL := TRUE;         // safety consent (TRUE = run)
    xPhotocell  : BOOL;                 // measure photocell: TRUE = covered by a piece
    xExitPhotocell : BOOL;              // exit photocell: TRUE = piece passing through (ejection)
    xResume     : BOOL;                 // external signal: reactivate the belt from standby
    xHomeBelt   : BOOL;                 // command: re-home the belt while stopped in standby

    // ── parameters (example defaults) ────────────────────────
    lrScale     : LREAL := 4096.0;      // counts/mm (same for both axes in this example)
    lrCycleTime : LREAL := 0.001;       // task period [s]
    lrBeltVel   : LREAL := 200.0;       // belt velocity [mm/s]
    lrBeltMove  : LREAL := 500.0;       // belt PP travel after the trigger [mm]
    lrThreshold : LREAL := 300.0;       // piece measurement threshold [mm]: > -> reject
    lrAx2Home   : LREAL := 10.0;        // reference position after axis2 homing [mm]
    lrAx2Fwd    : LREAL := 400.0;       // axis2 forward position (absolute) [mm]
    lrAx2Zero   : LREAL := 0.0;         // axis2 return position (absolute) [mm]
    lrAx2Vel    : LREAL := 150.0;       // axis2 velocity [mm/s]
    tWait       : TIME  := T#500MS;     // short waits
    tIdle       : TIME  := T#5S;        // timeout with no pieces -> belt standby
END_VAR
VAR_OUTPUT
    xHomed        : BOOL;               // homing complete (once)
    xRunning      : BOOL;               // cycle running
    xStandby      : BOOL;               // belt stopped in standby (no pieces)
    xError        : BOOL;               // error on an axis
    xOversize     : BOOL;               // last piece out of tolerance (rejected)
    xEjected      : BOOL;               // pulse: rejected piece certified ejected
    nRejectCount  : UINT;               // ejected-reject counter
    lrLastMeasure : LREAL;              // last piece measurement [mm]
    iStep         : INT;                // current step (diagnostics)
    sDiag         : STRING(80);         // diagnostic text of the axis in error
END_VAR
VAR CONSTANT
    nBelt : UINT := 1;                  // belt axis index in GVL_AXIS
    nPos  : UINT := 2;                  // positioner axis index in GVL_AXIS
END_VAR
VAR
    tonWait : TON;
    tonIdle : TON;
    ftExit  : F_TRIG;   // falling edge of the exit photocell
END_VAR

//------------------------------------------------------------------------------
// /axis configuration (every cycle)
//------------------------------------------------------------------------------
GVL_AXIS.Ctrl[nBelt].Scale          := lrScale;
GVL_AXIS.Ctrl[nBelt].CycleTime      := lrCycleTime;
GVL_AXIS.Ctrl[nBelt].xEmergencyStop := xSafetyOk;
GVL_AXIS.Ctrl[nPos].Scale           := lrScale;
GVL_AXIS.Ctrl[nPos].CycleTime       := lrCycleTime;
GVL_AXIS.Ctrl[nPos].xEmergencyStop  := xSafetyOk;

// short-wait timer (active only in steps 75/85, auto-resets on exit)
tonWait(IN := (iStep = 75) OR (iStep = 85), PT := tWait);
//------------------------------------------------------------------------------
// axis configuration/
//------------------------------------------------------------------------------

//------------------------------------------------------------------------------
// /supervision: error or stop while in the cycle
//------------------------------------------------------------------------------
xError := GVL_AXIS.State[nBelt].xError OR GVL_AXIS.State[nPos].xError;

IF    GVL_AXIS.State[nBelt].xError THEN sDiag := GVL_AXIS.State[nBelt].DiagText;
ELSIF GVL_AXIS.State[nPos].xError  THEN sDiag := GVL_AXIS.State[nPos].DiagText;
ELSE                                    sDiag := '';
END_IF

IF (iStep > 0) AND (iStep < 900) THEN
    IF    xError THEN iStep := 900;
    ELSIF xStop  THEN iStep := 200;
    END_IF
END_IF

xRunning := (iStep <> 0);
xStandby := (iStep >= 100) AND (iStep <= 120);
xEjected := FALSE;   // pulse: TRUE only when a piece passes the exit (parallel tracker)
//------------------------------------------------------------------------------
// supervision/
//------------------------------------------------------------------------------

//------------------------------------------------------------------------------
// /reject ejection — PARALLEL (the belt does not stop; a new piece can
//   already enter while the reject travels toward the exit photocell)
//------------------------------------------------------------------------------
ftExit(CLK := xExitPhotocell);
IF ftExit.Q THEN                    // falling edge = rejected piece passed = ejected
    nRejectCount := nRejectCount + 1;
    xEjected     := TRUE;           // pulse (1 scan)
END_IF

// standby timer: active only while waiting for a piece, reset when a piece arrives
tonIdle(IN := (iStep = 40) AND NOT xPhotocell, PT := tIdle);
//------------------------------------------------------------------------------
// reject ejection/
//------------------------------------------------------------------------------

CASE iStep OF

    0: // IDLE
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_NULL;
        GVL_AXIS.Ctrl[nPos].eCmd  := AXIS_NULL;
        IF xStart AND NOT xStop THEN iStep := 5; END_IF

    5: // ENABLE both axes
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_ENABLE;
        GVL_AXIS.Ctrl[nPos].eCmd  := AXIS_ENABLE;
        IF GVL_AXIS.Info[nBelt].xEnabled AND GVL_AXIS.Info[nPos].xEnabled THEN
            IF xHomed THEN iStep := 30; ELSE iStep := 10; END_IF   // homing once
        END_IF

    10: // HOMING axis1 (belt) — in place (drive method)
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_HOME;
        IF GVL_AXIS.State[nBelt].xDone THEN
            GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_NULL;
            iStep := 20;
        END_IF

    20: // HOMING axis2 — sensor + encoder index, reference = lrAx2Home
        GVL_AXIS.Ctrl[nPos].HomeOffset := LREAL_TO_DINT(lrAx2Home * lrScale);
        GVL_AXIS.Ctrl[nPos].eCmd       := AXIS_HOME;
        IF GVL_AXIS.State[nPos].xDone THEN
            GVL_AXIS.Ctrl[nPos].eCmd := AXIS_NULL;
            xHomed := TRUE;
            iStep  := 30;
        END_IF

    30: // BELT in velocity + arm TouchProbe (cycle entry / scanning)
        GVL_AXIS.Data[nBelt].lrVelocity          := lrBeltVel;
        GVL_AXIS.Ctrl[nBelt].eCmd                := AXIS_MOVE_VELOCITY;
        GVL_AXIS.Ctrl[nBelt].xTouchProbeEnable   := TRUE;
        GVL_AXIS.Ctrl[nBelt].xTouchProbeRising   := TRUE;
        GVL_AXIS.Ctrl[nBelt].xTouchProbeFalling  := TRUE;
        iStep := 40;

    40: // wait for a piece (photocell covered); if none for tIdle -> standby
        IF xPhotocell THEN
            // rising latched in HW; belt from velocity to relative PP, without stopping
            GVL_AXIS.Data[nBelt].lrTargetPosition := lrBeltMove;
            GVL_AXIS.Data[nBelt].lrVelocity       := lrBeltVel;
            GVL_AXIS.Ctrl[nBelt].eCmd             := AXIS_MOVE_REL;
            iStep := 50;
        ELSIF tonIdle.Q THEN
            iStep := 100;                          // no piece -> standby
        END_IF

    50: // wait for the piece to pass (photocell cleared) + measurement analysis
        IF NOT xPhotocell THEN
            lrLastMeasure := ABS(DINT_TO_LREAL(
                                 GVL_AXIS.Info[nBelt].TouchProbeFallingValue
                               - GVL_AXIS.Info[nBelt].TouchProbeRisingValue)) / lrScale;
            GVL_AXIS.Ctrl[nBelt].xTouchProbeEnable := FALSE;   // stop TouchProbe
            xOversize := (lrLastMeasure > lrThreshold);
            IF xOversize THEN
                // non-blocking REJECT: the belt returns to velocity and does NOT stop;
                // the piece will be ejected (it falls) and certified by the parallel
                // tracker on xExitPhotocell. Measuring resumes IMMEDIATELY (pipelining):
                // a new piece can enter while the reject exits.
                GVL_AXIS.Data[nBelt].lrVelocity := lrBeltVel;
                GVL_AXIS.Ctrl[nBelt].eCmd       := AXIS_MOVE_VELOCITY;
                iStep := 30;
            ELSE
                iStep := 70;          // good: stop and process with axis2
            END_IF
        END_IF

    70: // GOOD: wait for the belt (PP) to have stopped
        IF GVL_AXIS.State[nBelt].xDone THEN iStep := 75; END_IF

    75: // short wait, then axis2 FORWARD to lrAx2Fwd (absolute, e.g. +400)
        IF tonWait.Q THEN
            GVL_AXIS.Ctrl[nBelt].eCmd             := AXIS_NULL;
            GVL_AXIS.Data[nPos].lrTargetPosition  := lrAx2Fwd;     // forward to +400
            GVL_AXIS.Data[nPos].lrVelocity        := lrAx2Vel;
            GVL_AXIS.Ctrl[nPos].eCmd              := AXIS_MOVE_ABS;
            iStep := 80;
        END_IF

    80: // wait for axis2 at the forward position
        IF GVL_AXIS.State[nPos].xDone THEN iStep := 85; END_IF

    85: // short wait, then SIMULTANEOUSLY axis2 returns to 0 and belt in velocity
        IF tonWait.Q THEN
            GVL_AXIS.Data[nPos].lrTargetPosition := lrAx2Zero;     // absolute return to 0
            GVL_AXIS.Data[nPos].lrVelocity       := lrAx2Vel;
            GVL_AXIS.Ctrl[nPos].eCmd             := AXIS_MOVE_ABS;
            iStep := 30;              // restart belt (velocity + TP), axis2 in parallel
        END_IF

    100: // STANDBY: slow down and stop the belt (no piece for tIdle)
        GVL_AXIS.Ctrl[nBelt].eCmd              := AXIS_STOP;   // Halt = decelerate and hold
        GVL_AXIS.Ctrl[nBelt].xTouchProbeEnable := FALSE;
        IF NOT GVL_AXIS.Info[nBelt].xMoving THEN
            GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_NULL;
            iStep := 110;
        END_IF

    110: // STANDBY stopped: wait for home (re-zeroing) or a resume signal
        IF xHomeBelt THEN
            iStep := 120;                 // re-home the belt in place
        ELSIF xResume THEN
            iStep := 30;                  // reactivate rotation (exit standby)
        END_IF

    120: // STANDBY home: re-home the belt in place, then back to standby
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_HOME;
        IF GVL_AXIS.State[nBelt].xDone THEN
            GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_NULL;
            iStep := 110;
        END_IF

    200: // STOP requested — Halt both axes, then idle
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_STOP;
        GVL_AXIS.Ctrl[nPos].eCmd  := AXIS_STOP;
        IF (NOT GVL_AXIS.Info[nBelt].xMoving) AND (NOT GVL_AXIS.Info[nPos].xMoving) THEN
            GVL_AXIS.Ctrl[nBelt].eCmd              := AXIS_NULL;
            GVL_AXIS.Ctrl[nPos].eCmd               := AXIS_NULL;
            GVL_AXIS.Ctrl[nBelt].xTouchProbeEnable := FALSE;
            iStep := 0;
        END_IF

    900: // ERROR — reset both axes, then idle (needs a new xStart)
        GVL_AXIS.Ctrl[nBelt].eCmd := AXIS_RESET;
        GVL_AXIS.Ctrl[nPos].eCmd  := AXIS_RESET;
        IF NOT xError THEN
            GVL_AXIS.Ctrl[nBelt].eCmd              := AXIS_NULL;
            GVL_AXIS.Ctrl[nPos].eCmd               := AXIS_NULL;
            GVL_AXIS.Ctrl[nBelt].xTouchProbeEnable := FALSE;
            iStep := 0;
        END_IF

END_CASE
