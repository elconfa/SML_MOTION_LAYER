(* ============================================================
   GVL_IO — Immagini di processo del bridge I/O (hardware)
   SML v9 — bridge I/O

   Usato SOLO col bridge. Su CoDeSys con link diretto NON serve:
   in quel caso lascia IO_LINK_ENABLE = FALSE (i due SML_IoLink_*
   si auto-bypassano) e mappa GVL_AXIS.Axis[n] direttamente.

   MAPPATURA (importante): DriveOut/DriveIn sono array SEMPLICI, senza
   'AT %Q*'/'AT %I*'. Motivo: su CoDeSys un indirizzo incompleto in un
   GVL richiede un VAR_CONFIG/mappatura e altrimenti da' errore C0128.
   Con array semplici il codice compila standalone; i canali PDO si
   collegano cosi':
     - CoDeSys: scheda <Slave> I/O Mapping -> colonna Variable ->
       lega ogni PDO a GVL_IO.DriveOut[n].<campo> / DriveIn[n].<campo>.
     - TwinCAT: 'Change Link' su ogni PDO verso GVL_IO.DriveOut/DriveIn,
       oppure (se preferisci l'immagine di processo esplicita) riaggiungi
       'AT %Q*' / 'AT %I*' — supportato su TwinCAT.
   Vedi docs/GUIDA_IO_Linking.md.
   ============================================================ *)

{attribute 'qualified_only'}
VAR_GLOBAL
    // flag di abilitazione del bridge (design-time o runtime)
    IO_LINK_ENABLE : BOOL := FALSE;   // TRUE = attiva SML_IoLink_In / _Out

    DriveOut : ARRAY[1..GVL_SML_CONST.MAX_AXIS] OF ST_DriveOut;  // -> RxPDO (mappa nel configuratore)
    DriveIn  : ARRAY[1..GVL_SML_CONST.MAX_AXIS] OF ST_DriveIn;   // <- TxPDO (mappa nel configuratore)
END_VAR
