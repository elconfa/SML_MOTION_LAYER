(* ============================================================
   SML_ProfilePosition — CiA402 Profile Position Mode (PP, mode 1)
   SML v4 — Merged (OpenSML_ProfilePosition + SML_ProfilePosition/Confa)

   Profile Acceleration and Deceleration are passed per-call and
   written to Axis.Profile_Acceleration / Axis.Profile_Deceleration
   (0x6083 / 0x6084) via PDO — no SDO access required.

   CiA402 PP setpoint protocol (implemented via states 10→20→30→40):
     1. Write Target_Position, Profile_Velocity, Accel, Decel
     2. Set ControlWord bit 4 (New Setpoint) = 1
     3. Wait StatusWord bit 12 (Setpoint Acknowledge) = 1
     4. Clear ControlWord bit 4 = 0
     5. Wait StatusWord bit 10 (Target Reached) = 1

   States:
     0   Idle — wait for enable + execute
     10  Mode Switch — switch to PP mode 1
     20  Setpoint active — wait for acknowledge / target / error
     30  Acknowledged — wait for target reached or new execute
     40  Target reached — hold xDone until execute drops
     999 Error — hold position; reset by xReset or NOT xEnable
   ============================================================ *)

FUNCTION_BLOCK SML_ProfilePosition
VAR_IN_OUT
    Axis : OpenSML_Axis;
END_VAR
VAR_INPUT
    xEnable                : BOOL;   // Level: keep FB active
    xExecute               : BOOL;   // Trigger: start move (hold for duration)
    xContinueMove          : BOOL;   // Blend: no deceleration between consecutive moves (bit 5)
    xRelativeMode          : BOOL;   // FALSE = absolute, TRUE = relative (bit 6)
    xReset                 : BOOL;   // Pulse: acknowledge error
    diTargetPosition       : DINT;   // Target [encoder counts, raw drive units]
    udiProfileVelocity     : UDINT;  // Move speed [drive velocity units]; must be > 0
    udiProfileAcceleration : UDINT;  // Acceleration [drive accel units]
    udiProfileDeceleration : UDINT;  // Deceleration [drive accel units]
END_VAR
VAR_OUTPUT
    xDone   : BOOL; // Target position reached (bit 10 set, no following error)
    xBusy   : BOOL; // Setpoint acknowledged (bit 12), motion in progress
    xMoving : BOOL; // Axis velocity ≠ 0 (StatusWord bit 14 = 0 on many drives)
    xError  : BOOL; // Fault (bit 3) or following error (bit 13)
    eProgress : E_PROGRESS; // Livello A: avanzamento unificato (proiezione di iState)
END_VAR
VAR
    iState : INT;
END_VAR

// ── Continuous outputs (updated every cycle while in PP mode) ─────────────
// xBusy: setpoint acknowledged AND target not yet reached
IF xEnable AND Axis.Modes_of_operation_display = 1 THEN
    xBusy := (iState > 0 AND iState < 999)
             AND Axis.StatusWord.12    // Setpoint Acknowledge
             AND NOT Axis.StatusWord.10; // NOT Target Reached
END_IF

// xMoving: bit 14 = standstill flag (0 = moving) — drive-dependent
xMoving := NOT Axis.StatusWord.14;

// ── Write ramp parameters only when commanded ─────────────────────────────
IF xExecute THEN
    Axis.Profile_Acceleration := udiProfileAcceleration;
    Axis.Profile_Deceleration := udiProfileDeceleration;
END_IF

CASE iState OF

    0: // Idle
        xDone  := FALSE;
        xError := FALSE;

        IF xEnable THEN
            IF xExecute AND Axis.StatusWord.1 AND NOT Axis.StatusWord.3 THEN
                // Drive is at least SwitchedOn and no fault — attempt move
                Axis.ControlWord.5    := xContinueMove;
                Axis.ControlWord.6    := xRelativeMode;
                Axis.Profile_Velocity := udiProfileVelocity;
                Axis.Target_Position  := diTargetPosition;
                iState := 10;
            ELSIF xExecute OR Axis.StatusWord.3 THEN
                iState := 999;
            END_IF
        END_IF

    10: // Mode Switch — request PP (mode 1), re-enable operation when confirmed
        Axis.Modes_of_operation := 1;
        Axis.ControlWord.3 := (Axis.Modes_of_operation_display = 1);

        IF Axis.StatusWord.3 OR NOT xEnable THEN
            iState := 999;
        ELSIF Axis.StatusWord.2 AND Axis.Modes_of_operation_display = 1 THEN
            // OperationEnabled in PP mode → trigger new setpoint (rising edge on bit 4)
            Axis.ControlWord.4 := TRUE;
            iState := 20;
        END_IF

    20: // Active — setpoint sent, waiting for acknowledge/target/error
        IF Axis.StatusWord.13 OR Axis.StatusWord.3 OR NOT xEnable THEN
            // Following error, fault, or operator cancel
            iState := 999;
        ELSIF Axis.StatusWord.10 THEN
            // Target Reached (bit 10) — move complete
            xDone := TRUE;
            Axis.ControlWord.4 := FALSE;
            iState := 40;
        ELSIF Axis.StatusWord.12 AND NOT xExecute THEN
            // Setpoint acknowledged but xExecute already dropped
            // → clear bit 4 and move to ready-for-next state
            Axis.ControlWord.4 := FALSE;
            iState := 30;
        ELSIF NOT Axis.StatusWord.12 AND NOT Axis.StatusWord.10 AND NOT xExecute THEN
            // Neither acknowledged nor reached, execute dropped → clean exit
            Axis.ControlWord.4 := FALSE;
            iState := 0;
        END_IF

    30: // Acknowledged — moving toward target; accept next execute or wait
        IF Axis.StatusWord.13 OR Axis.StatusWord.3 OR NOT xEnable THEN
            iState := 999;
        ELSIF Axis.StatusWord.10 THEN
            xDone := TRUE;
            Axis.ControlWord.4 := FALSE;
            iState := 40;
        ELSIF xExecute AND Axis.StatusWord.1 AND NOT Axis.StatusWord.3 THEN
            // New move requested — queue it
            Axis.ControlWord.5    := xContinueMove;
            Axis.ControlWord.6    := xRelativeMode;
            Axis.Profile_Velocity := udiProfileVelocity;
            Axis.Target_Position  := diTargetPosition;
            iState := 10;
        END_IF

    40: // Target reached — hold xDone until execute or enable drops
        IF NOT xEnable OR NOT xExecute THEN
            xDone  := FALSE;
            iState := 0;
        END_IF

    999: // Error — hold current position; reset by xReset or dropping xEnable
        Axis.Profile_Velocity := 0;
        // SAFETY: hold current position, never write 0 (that commands a move to origin)
        Axis.Target_Position  := Axis.Position_Actual_Value;
        Axis.ControlWord.4    := FALSE;
        xError := TRUE;
        IF NOT xEnable OR xReset THEN
            xError := FALSE;
            iState := 0;
        END_IF

END_CASE

// ── Livello A: proiezione avanzamento unificato (E_PROGRESS) ──────────────
IF    xError     THEN eProgress := PROGRESS_ERROR;
ELSIF xDone      THEN eProgress := PROGRESS_DONE;
ELSIF iState = 0 THEN eProgress := PROGRESS_INVALID;
ELSE                  eProgress := PROGRESS_BUSY;
END_IF
