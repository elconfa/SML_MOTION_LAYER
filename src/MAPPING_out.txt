(* ============================================================
   MAPPING_out — Copia stato/info da GVL_AXIS verso il bus
   SML v8 — Livello B / MAPPING (nuovo)

   Percorso: GVL_AXIS.State/Info  ->  GVL_AXIS_MAP.State/Info (fieldbus)
   Chiamare alla FINE di MAIN, dopo il loop assi.

   GATE: attivo solo se GVL_AXIS_MAP.AXIS_MAP_ENABLE = TRUE; altrimenti
   bypass (flag runtime; vedi nota in MAPPING_in).
   ============================================================ *)

PROGRAM MAPPING_out
VAR
    i       : UINT;
    bSizeOk : BOOL;
END_VAR

IF NOT GVL_AXIS_MAP.AXIS_MAP_ENABLE THEN RETURN; END_IF

// ── Check dimensione blocchi bus (sicurezza) ─────────────────
bSizeOk := (SIZEOF(ST_AXIS_STATE) <= GVL_SML_CONST.MAP_SIZE_STATE)
       AND (SIZEOF(ST_AXIS_INFO)  <= GVL_SML_CONST.MAP_SIZE_INFO);
IF NOT bSizeOk THEN RETURN; END_IF   // blocco: ridimensionare MAP_SIZE_*

// ── Copia interno -> bus ─────────────────────────────────────
FOR i := 1 TO GVL_SML_CONST.MAX_AXIS DO
    GVL_AXIS_MAP.State[i].stData := GVL_AXIS.State[i];
    GVL_AXIS_MAP.Info[i].stData  := GVL_AXIS.Info[i];
END_FOR
