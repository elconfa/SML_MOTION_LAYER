(* ============================================================
   ST_AXIS_CTRL — Struttura di comando (app -> layer)
   SML v6 — Livello B (nuovo)

   Contratto dati "Ctrl" del pattern Motion Layer. Struttura piatta
   e serializzabile (nessun puntatore/interfaccia dentro): accessibile
   via simbolo ADS o copiabile su bus (predisposizione MAPPING).

   Il comando eCmd e' level-based. I parametri di moto stanno in
   ST_MOVE_DATA ("Data").
   ============================================================ *)

TYPE ST_AXIS_CTRL :
STRUCT
    eCmd           : E_AXIS_CTRL;        // comando richiesto
    xEmergencyStop : BOOL := TRUE;       // TRUE = run / FALSE = quick-stop (CiA402 bit 2)
    xSimulation    : BOOL;               // specchia Target_Position -> Position_Actual_Value

    // ── configurazione (raramente variabile) ────────────────
    Scale          : LREAL := 4096.0;    // [counts/unita'] — deve essere > 0
    CycleTime      : LREAL := 0.001;     // [s] — deve combaciare col task (OTG)
    PositionWindow : LREAL := 0.01;      // deadband in-posizione [unita']
    HomeOffset     : DINT  := 0;         // offset homing [encoder counts] -> 0x607C

    // ── TouchProbe (parallelo al moto, non e' un eCmd) ───────
    xTouchProbeEnable  : BOOL;           // arma l'hardware TouchProbe
    xTouchProbeRising  : BOOL;           // arma il latch sul fronte di salita
    xTouchProbeFalling : BOOL;           // arma il latch sul fronte di discesa
END_STRUCT
END_TYPE
