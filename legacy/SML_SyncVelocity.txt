(* ============================================================
   SML_SyncVelocity - CSP-based Cyclic Synchronous Velocity
   SML v4 - Merged (OpenSML_SyncVelocity fixed + Confa improvements)

   v4 fixes:
   - C1: otg called with correct param name CycleTIme (v3 used CycleTime ->
     did not compile).
   - M1: explicit OTG seed via xInit.
   - M2: simulation path now passes through the seed state (state 10).
   - B4: simplified signed LREAL_TO_DINT conversion.

   Integrates velocity in PLC via OTG, writes Target_Position
   (CSP mode 8) every cycle - avoids requiring a CSV PDO mapping.

   If the drive supports CSV (mode 9) natively:
     Uncomment the otg.NewVelocity line below and switch mode to 9.

   REQUIRES: StruckigLight library (FB_S7RTT_OTG)
   ============================================================ *)

FUNCTION_BLOCK SML_SyncVelocity
VAR_IN_OUT
    Axis : OpenSML_Axis;
END_VAR
VAR_INPUT
    xEnable         : BOOL;
    xSimulation     : BOOL;
    TargetVelocity  : LREAL;          // [user units/s]; signed; 0 = decelerate to stop
    MaxVelocity     : LREAL := 2000.0;
    MaxAcceleration : LREAL := 200.0;
    MaxJerk         : LREAL := 200.0;
    CycleTime       : LREAL := 0.001;
    lrScale         : LREAL := 1.0;   // [counts/unit]; must be > 0
END_VAR
VAR_OUTPUT
    xBusy  : BOOL;
    xError : BOOL;
    eProgress : E_PROGRESS; // Level A: unified progress (projection of iState)
END_VAR
VAR
    otg    : FB_S7RTT_OTG;
    iState : INT;
END_VAR

IF lrScale = 0.0 THEN lrScale := 1.0; END_IF
xBusy := (iState = 10) OR (iState = 20);

CASE iState OF
    0:
        xError := FALSE;
        IF xEnable THEN
            IF Axis.StatusWord.1 AND Axis.StatusWord.2
               AND Axis.Modes_of_operation_display = 8
               AND NOT Axis.StatusWord.3 THEN
                iState := 10;
            ELSIF Axis.StatusWord.1 AND Axis.Modes_of_operation_display <> 8 THEN
                Axis.ControlWord.3 := FALSE;
                iState := 10;
            ELSIF xSimulation THEN
                iState := 10; // M2: still seed the OTG via the mode-switch state
            ELSE
                iState := 999;
            END_IF
        END_IF

    10: // Mode Switch to CSP (8) - used as velocity integrator; also seeds OTG
        // M1: call OTG here with xInit=TRUE so the integrator is (re)seeded to
        // the actual position every cycle while in this state.
        Axis.Modes_of_operation := 8;
        Axis.ControlWord.3 := (Axis.Modes_of_operation_display = 8);
        otg.CurrentPosition := DINT_TO_LREAL(Axis.Position_Actual_Value) / lrScale;
        otg(xInit            := TRUE,
            ControlInterface := TRUE,
            TargetPosition   := 0.0,
            TargetVelocity   := 0.0,
            CycleTIme        := CycleTime,
            MaxVelocity      := MaxVelocity,
            MaxAcceleration  := MaxAcceleration,
            MaxJerk          := MaxJerk);
        IF (Axis.StatusWord.2 AND Axis.Modes_of_operation_display = 8) OR xSimulation THEN
            iState := 20;
        END_IF

    20: // OTG velocity integration loop
        IF xSimulation THEN
            Axis.Position_Actual_Value := Axis.Target_Position;
        END_IF
        otg(
            xInit            := FALSE,
            ControlInterface := TRUE,          // Velocity interface: OTG integrates to position
            TargetPosition   := 0.0,
            TargetVelocity   := TargetVelocity,
            CycleTIme        := CycleTime,
            MaxVelocity      := MaxVelocity,
            MaxAcceleration  := MaxAcceleration,
            MaxJerk          := MaxJerk);

        Axis.Target_Position := LREAL_TO_DINT(otg.NewPosition * lrScale);
        // For native CSV (mode 9), also write:
        // Axis.Target_Velocity := LREAL_TO_DINT(otg.NewVelocity * lrScale);

        IF Axis.StatusWord.13 OR Axis.StatusWord.3 OR otg.xError THEN
            iState := 999;
        ELSIF NOT xEnable THEN
            iState := 0;
        END_IF

    999:
        xError := TRUE;
        IF NOT xEnable THEN
            xError := FALSE;
            iState := 0;
        END_IF
END_CASE

// -- Level A: unified progress projection (E_PROGRESS) --------------
// Continuous sync: no discrete "DONE".
IF    xError     THEN eProgress := PROGRESS_ERROR;
ELSIF iState = 0 THEN eProgress := PROGRESS_INVALID;
ELSE                  eProgress := PROGRESS_BUSY;
END_IF
