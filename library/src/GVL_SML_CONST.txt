    // by FB_Mapping_In/Out via Check). Fixed for a stable process image
    // on the fieldbus.
    MAP_SIZE_CTRL  : UDINT := 64;    // >= SIZEOF(ST_AXIS_CTRL)
    MAP_SIZE_DATA  : UDINT := 64;    // >= SIZEOF(ST_MOVE_DATA)
    MAP_SIZE_STATE : UDINT := 256;   // >= SIZEOF(ST_AXIS_STATE) (holds 2x STRING(80))
    MAP_SIZE_INFO  : UDINT := 128;   // >= SIZEOF(ST_AXIS_INFO) (with Status + TouchProbe)
END_VAR
