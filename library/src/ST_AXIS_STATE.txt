(* ============================================================
   ST_AXIS_STATE - State structure (layer -> app)
   SML v6 - Level B (new)

   The "State" data contract. Command result + progress.
   eState is the COMBINED STATE (E_AXIS_STATE + E_PROGRESS);
   use f_GetState / f_GetProgress to decompose it.
   ============================================================ *)

TYPE ST_AXIS_STATE :
STRUCT
    eState         : INT;            // combined state = E_AXIS_STATE + E_PROGRESS
    eProgress      : E_PROGRESS;     // progress (handy, already decomposed)
    eCmdActive     : E_AXIS_CTRL;    // command being executed
    xDone          : BOOL;           // current command complete
    xError         : BOOL;           // aggregated error

    // -- diagnostics (from SML_Diagnostics) -------------------
    DiagCode       : SML_DiagCode;
    DiagText       : STRING(80);
    FirstFaultCode : SML_DiagCode;
    FirstFaultText : STRING(80);
END_STRUCT
END_TYPE
