(* ============================================================
   ST_CiA402_Status — Decoder di stato CiA402 completo
   SML v9 — Livello B (nuovo)

   Replica gli output del decoder SML_Status, resi disponibili nel
   contratto dati come sotto-struct di ST_AXIS_INFO (Info.Status).
   Porta in FB_AxisCtrl i bit di stato granulari che prima erano
   esposti solo da FB_SML.
   ============================================================ *)

TYPE ST_CiA402_Status :
STRUCT
    // ── macchina di stato CiA402 ─────────────────────────────
    NotReadyToSwitchOn  : BOOL;
    SwitchOnDisabled    : BOOL;
    ReadyToSwitchOn     : BOOL;
    SwitchedOn          : BOOL;
    OperationEnabled    : BOOL;
    QuickStopActive     : BOOL;
    FaultReactionActive : BOOL;
    Fault               : BOOL;

    // ── stato generale ───────────────────────────────────────
    Warning             : BOOL;   // bit 7
    TargetReached       : BOOL;   // bit 10 (mode-dependent)
    Halted              : BOOL;
    VoltageEnabled      : BOOL;   // bit 4
    InternalLimit       : BOOL;   // bit 11

    // ── modo operativo attivo ────────────────────────────────
    ProfilePositionMode : BOOL;   // 1
    ProfileVelocityMode : BOOL;   // 3
    VelocityMode        : BOOL;   // 2 (legacy)
    TorqueProfileMode   : BOOL;   // 4
    HomingMode          : BOOL;   // 6
    SyncPositionMode    : BOOL;   // 8 (CSP)
    SyncVelocityMode    : BOOL;   // 9 (CSV)

    // ── moto ─────────────────────────────────────────────────
    Moving              : BOOL;
    StandStill          : BOOL;
    Homed               : BOOL;

    // ── errore / dettaglio ───────────────────────────────────
    Err                 : BOOL;   // Error_Code != 0
    ErrId               : UINT;   // 0x603F
    ActTorque           : INT;    // 0x6077 [0.1% del nominale]
    FollowingError      : DINT;   // 0x60F4 [encoder counts]
END_STRUCT
END_TYPE
