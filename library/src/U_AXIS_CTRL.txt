(* ============================================================
   U_AXIS_CTRL — Bus UNION for the axis command
   SML v8 — Level B / MAPPING (new)

   Overlay of the same memory block as:
     - stData : ST_AXIS_CTRL   (typed view for the logic)
     - aRaw   : byte block      (raw view for fieldbus mapping)

   Fixed size = MAP_SIZE_CTRL (>= SIZEOF(ST_AXIS_CTRL), checked by
   PRG_Mapping_In). Lets you link the element to the process image as
   a struct or as a raw block without changing the mapped size.
   ============================================================ *)

TYPE U_AXIS_CTRL :
UNION
    stData : ST_AXIS_CTRL;
    aRaw   : ARRAY[1..GVL_SML_CONST.MAP_SIZE_CTRL] OF BYTE;
END_UNION
END_TYPE
