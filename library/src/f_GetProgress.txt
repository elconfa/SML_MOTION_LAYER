(* ============================================================
   f_GetProgress — Extracts the progress from a combined state
   SML v5 — Level A (new)

   eCombined = functional_state + E_PROGRESS  (progress band = 0..99)
   Returns only the progress part.

   Level B groundwork: not used by the Level A projection, provided
   as infrastructure for the combined state.
   ============================================================ *)

FUNCTION f_GetProgress : E_PROGRESS
VAR_INPUT
    eCombined : DINT;   // combined state = functional_state + progress
END_VAR

f_GetProgress := TO_INT(eCombined MOD GVL_SML_CONST.PROGRESS_SPAN);
