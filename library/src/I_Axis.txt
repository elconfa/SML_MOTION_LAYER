(* ============================================================
   I_Axis - Axis interface (equivalent of I_McAxis)
   SML v6 - Level B (new)

   A method facade to drive an axis without depending on the
   implementation. It serves high-level coordinators (in the
   future: CAM/NCI/gearing) to maneuver the axes.

   SEMANTICS: the methods are SETTERS. They set Ctrl.eCmd + Data and
   return the current progress (State.eProgress). The actual
   execution happens in the cyclic body of FB_AxisCtrl, which MUST
   be called every cycle. Use EITHER the methods OR the Ctrl struct,
   not both at the same time on the same axis.

   Textual representation: in CoDeSys each METHOD/PROPERTY listed
   here is a separate object under the INTERFACE I_Axis.
   ============================================================ *)

INTERFACE I_Axis

// -- METHODS (return E_PROGRESS) --------------------------------

METHOD Enable : E_PROGRESS
// (no parameters)

METHOD Disable : E_PROGRESS
// (no parameters)

METHOD Reset : E_PROGRESS
// (no parameters)

METHOD Home : E_PROGRESS
// (no parameters; uses Ctrl.HomeOffset)

METHOD MoveAbsolute : E_PROGRESS
VAR_INPUT
    lrPosition : LREAL;   // [units]
    lrVelocity : LREAL;   // [units/s]
END_VAR

METHOD MoveRelative : E_PROGRESS
VAR_INPUT
    lrDistance : LREAL;   // [units]
    lrVelocity : LREAL;   // [units/s]
END_VAR

METHOD MoveVelocity : E_PROGRESS
VAR_INPUT
    lrVelocity : LREAL;   // [units/s] signed
END_VAR

METHOD Jog : E_PROGRESS
VAR_INPUT
    xForward   : BOOL;
    xReverse   : BOOL;
    lrVelocity : LREAL;   // magnitude [units/s]
END_VAR

METHOD MoveFollow : E_PROGRESS   // CSP + OTG (jerk-limited)
VAR_INPUT
    lrPosition : LREAL;   // [units] - online re-targetable
    lrVelocity : LREAL;   // [units/s]
END_VAR

METHOD Stop : E_PROGRESS
// (no parameters)

// -- PROPERTIES (get) -------------------------------------------

PROPERTY Position : LREAL     // actual position [units]  (get)
PROPERTY Enabled  : BOOL      // drive OperationEnabled     (get)

END_INTERFACE
