(* ============================================================
   FB_AxisCtrl — Methods and properties of I_Axis
   SML v6 — Level B (new)

   REAL code (not commented out) of the methods/properties that
   FB_AxisCtrl must implement to satisfy I_Axis.
   In CoDeSys: create a METHOD / PROPERTY object under FB_AxisCtrl
   for each block below.

   They are SETTERS: they set Ctrl.eCmd (+ Data) and return the
   current State.eProgress; motion happens on the next cyclic call
   of the FB body.
   ============================================================ *)

//============================================================ METHODS
METHOD Enable : E_PROGRESS
Ctrl.eCmd := AXIS_ENABLE;
Enable    := State.eProgress;

//------------------------------------------------------------
METHOD Disable : E_PROGRESS
Ctrl.eCmd := AXIS_DISABLE;
Disable   := State.eProgress;

//------------------------------------------------------------
METHOD Reset : E_PROGRESS
Ctrl.eCmd := AXIS_RESET;
Reset     := State.eProgress;

//------------------------------------------------------------
METHOD Home : E_PROGRESS
Ctrl.eCmd := AXIS_HOME;
Home      := State.eProgress;

//------------------------------------------------------------
METHOD MoveAbsolute : E_PROGRESS
VAR_INPUT
    lrPosition : LREAL;
    lrVelocity : LREAL;
END_VAR
Ctrl.eCmd             := AXIS_MOVE_ABS;
Data.lrTargetPosition := lrPosition;
Data.lrVelocity       := lrVelocity;
MoveAbsolute          := State.eProgress;

//------------------------------------------------------------
METHOD MoveRelative : E_PROGRESS
VAR_INPUT
    lrDistance : LREAL;
    lrVelocity : LREAL;
END_VAR
Ctrl.eCmd             := AXIS_MOVE_REL;
Data.lrTargetPosition := lrDistance;
Data.lrVelocity       := lrVelocity;
MoveRelative          := State.eProgress;

//------------------------------------------------------------
METHOD MoveVelocity : E_PROGRESS
VAR_INPUT
    lrVelocity : LREAL;   // signed
END_VAR
Ctrl.eCmd       := AXIS_MOVE_VELOCITY;
Data.lrVelocity := lrVelocity;
MoveVelocity    := State.eProgress;

//------------------------------------------------------------
METHOD Jog : E_PROGRESS
VAR_INPUT
    xForward   : BOOL;
    xReverse   : BOOL;
    lrVelocity : LREAL;   // magnitudo
END_VAR
IF    xForward THEN Ctrl.eCmd := AXIS_JOG_POS;
ELSIF xReverse THEN Ctrl.eCmd := AXIS_JOG_NEG;
ELSE               Ctrl.eCmd := AXIS_NULL;
END_IF
Data.lrVelocity := lrVelocity;
Jog             := State.eProgress;

//------------------------------------------------------------
METHOD MoveFollow : E_PROGRESS   // CSP + OTG
VAR_INPUT
    lrPosition : LREAL;
    lrVelocity : LREAL;
END_VAR
Ctrl.eCmd             := AXIS_MOVE_CSP;
Data.lrTargetPosition := lrPosition;
Data.lrVelocity       := lrVelocity;
MoveFollow            := State.eProgress;

//------------------------------------------------------------
METHOD Stop : E_PROGRESS
Ctrl.eCmd := AXIS_STOP;
Stop      := State.eProgress;

//============================================================ PROPERTIES
PROPERTY Position : LREAL
// GET:
Position := Info.lrActualPosition;

//------------------------------------------------------------
PROPERTY Enabled : BOOL
// GET:
Enabled := Info.xEnabled;
